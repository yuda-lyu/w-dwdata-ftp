_optReader.mjs

import isfun from 'wsemi/src/isfun.mjs'


//本檔為套件內部使用, 不由WDwdataFtp匯出
//

//warned, 已警告之項目, 同一函數、同一鍵每個行程只警告一次(dspec輸入參數之檢測機制K3)
let warned = new Set()


//readKey, 讀opt之鍵, opt非物件或讀取拋錯(存取器、已撤銷之Proxy)皆視為未給
let readKey = (opt, key) => {
    try {
        if ((typeof opt === 'object' && opt !== null) || typeof opt === 'function') {
            return opt[key]
        }
    }
    catch (err) {}
    return undefined
}


//emit, 輸出警告, 出口為opt.funLog(函數, 以(msg, { fn, reason, key })呼叫), 未給則為console.warn; 出口拋錯不影響流程
let emit = (funLog, msg, info) => {
    try {
        if (funLog !== null) {
            funLog(msg, info)
        }
        else {
            console.warn(msg)
        }
    }
    catch (err) {}
}


/**
 * 取opt之各鍵並檢測(規格: 輸入參數之檢測機制K3、V5、V11), 供downloadFiles與WDwdataFtp共用
 *
 * 規則:
 *   opt與各鍵為undefined或null視為未給, 取預設
 *   給了而檢測函數判為非預期者視為未給, 改用預設, 不報錯
 *   每個鍵只讀一次, 讀取拋錯(存取器、已撤銷之Proxy)視為未給
 *   opt.debug為true(非布林視為false)時, 非預期另輸出警告, 出口為opt.funLog(函數)或console.warn;
 *     警告為固定詞彙, 只含函數名、鍵名、種類描述與所採之預設(程式常數), 不含傳入之值(K9); 同一函數同一鍵每個行程只警告一次
 *   未宣告之鍵不檢查(opt常整份轉傳, V7)
 *
 * wsemi(1.9.79)僅於內部提供同類之opt解析, 未公開, 為免引用內部模組, 本套件以此最小實作暫代, 待wsemi公開後改用
 *
 * @param {String} fn 輸入函數名字串, 用於警告之前綴
 * @param {Object} [opt] 輸入呼叫端之opt, 非物件視為{}
 * @returns {Function} 回傳讀取函數(key, def, check, desc, defDesc), check為檢測函數(回布林, 拋錯視為非預期), desc為種類描述(如'a boolean'), defDesc為預設值描述(程式常數字串); 函數帶屬性debug(布林)與funLog(函數或null, 僅debug為true時有值), 為本次所讀之快照
 */
let optReader = (fn, opt) => {

    //debug, 僅恰為true才啟用
    let debug = readKey(opt, 'debug') === true

    //funLog
    let funLog = null
    if (debug) {
        let f = readKey(opt, 'funLog')
        funLog = isfun(f) ? f : null
    }

    //rd
    let rd = (key, def, check, desc, defDesc) => {

        //v
        let v = readKey(opt, key)
        if (v === undefined || v === null) {
            return def
        }

        //check
        let b = false
        try {
            b = check(v) === true
        }
        catch (err) {}
        if (b) {
            return v
        }

        //warn
        if (debug) {
            let k = `${fn}.${key}`
            if (!warned.has(k)) {
                warned.add(k)
                emit(funLog, `[w-dwdata-ftp ${fn}] invalid opt.${key}: must be ${desc}, using default ${defDesc}`, { fn, reason: 'invalid', key })
            }
        }

        return def
    }

    //debug與funLog僅讀取一次, 供轉傳給下層函數之呼叫端沿用(不再讀opt)
    rd.debug = debug
    rd.funLog = funLog

    return rd
}


//readKey, 讀opt之鍵(讀取拋錯視為未給), 供轉傳之鍵使用
optReader.readKey = readKey


//resetWarned, 重設已警告之紀錄, 僅供測試使用
optReader.resetWarned = () => {
    warned.clear()
}


export default optReader
